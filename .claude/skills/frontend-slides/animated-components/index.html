// split into words, wrap each in overflow:hidden span, stagger transitionDelay
function mountKinetic(el){
  const words = el.dataset.text.split(' ');
  el.innerHTML = words.map(w => `<span class="word"><span>${w}&nbsp;</span></span>`).join('');
  el.querySelectorAll('.word span').forEach((s,i)=> s.style.transitionDelay = `${i*70}ms`);
  requestAnimationFrame(()=> requestAnimationFrame(()=> el.classList.add('play')));
}

function animateMetric(target, ringEl, valueEl, circumference=402){
  const t0 = performance.now(), dur = 1600;
  ringEl.style.strokeDashoffset = circumference;
  function tick(t){
    const p = Math.min((t - t0) / dur, 1);
    const eased = 1 - Math.pow(1 - p, 3); // ease-out-cubic
    valueEl.textContent = Math.round(target * eased) + '%';
    ringEl.style.strokeDashoffset = circumference * (1 - eased);
    if (p < 1) requestAnimationFrame(tick);
  }
  requestAnimationFrame(tick);
}

function playTimeline(root){
  const steps = root.querySelectorAll('.tl-step');
  steps.forEach((s,i)=>{
    setTimeout(()=>{
      s.querySelector('.tl-node').classList.add('lit');
      const fill = s.querySelector('.tl-connector .fill');
      if (fill) fill.style.width = '100%';
    }, i * 450);
  });
}

card.addEventListener('mousemove', (e) => {
  const r = card.getBoundingClientRect();
  const x = (e.clientX - r.left) / r.width - 0.5;
  const y = (e.clientY - r.top) / r.height - 0.5;
  card.style.transform =
    `rotateY(${x * 10}deg) rotateX(${-y * 10}deg) translateZ(6px)`;
  card.querySelector('.tilt-glare').style.background =
    `radial-gradient(circle at ${ (x+0.5)*100 }% ${ (y+0.5)*100 }%, rgba(223,255,79,0.18), transparent 55%)`;
});
card.addEventListener('mouseleave', () => { card.style.transform = 'rotateY(0) rotateX(0)'; });

function playMindmap(root){
  const paths = root.querySelectorAll('.mm-path');
  const nodes = root.querySelectorAll('.mm-node');
  paths.forEach((p,i)=> setTimeout(()=> p.classList.add('drawn'), i*140));
  nodes.forEach((n,i)=> setTimeout(()=> n.classList.add('bloom'), 200 + i*140));
}
// SVG paths use stroke-dasharray/dashoffset for the draw-in;
// .bloom flips node opacity 0→1 and transform scale(.5)→scale(1)

.mesh-blob { position:absolute; border-radius:50%; filter:blur(50px); opacity:.55;
  animation: drift 16s cubic-bezier(.16,1,.3,1) infinite alternate; }
@keyframes drift {
  0% { transform: translate(0,0) scale(1); }
  100% { transform: translate(var(--dx,40px), var(--dy,30px)) scale(1.15); }
}
// "Shuffle" just rerolls each blob's --dx/--dy custom properties
// so the same CSS keyframe drifts toward a new random point.
document.querySelectorAll('.mesh-blob').forEach(b => {
  b.style.setProperty('--dx', `${(Math.random()*80-40).toFixed(0)}px`);
  b.style.setProperty('--dy', `${(Math.random()*80-40).toFixed(0)}px`);
});
